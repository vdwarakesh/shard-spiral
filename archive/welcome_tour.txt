    isMobile: () => window.matchMedia && window.matchMedia('(max-width: 900px)').matches,

    setDsaModal: (open) => {
    const el = document.getElementById('dsa-modal-backdrop');
    if (el) el.classList.toggle('open', Boolean(open));
    },

    setMobileHistoryModal: (open) => {
    const el = document.getElementById('mobile-history-modal-backdrop');
    if (el) el.classList.toggle('open', Boolean(open));
    },

    expandHeaderGroup: (side, targetId) => {
    const tb = document.querySelector('.top-titlebar');
    if (!tb) return;
    if (side === 'left') {
        tb.classList.add('expand-left');
        tb.classList.remove('expand-right');
    } else if (side === 'right') {
        tb.classList.add('expand-right');
        tb.classList.remove('expand-left');
    }
    if (targetId) {
        const targetEl = document.getElementById(targetId);
        if (targetEl && typeof targetEl.scrollIntoView === 'function') {
        targetEl.scrollIntoView({ inline: 'center', block: 'nearest' });
        }
    }
    },

    scrollCardIntoModalView: (id) => {
    const el = document.getElementById(id);
    if (el && typeof el.scrollIntoView === 'function') {
        el.scrollIntoView({ block: 'nearest', inline: 'nearest' });
    }
    }
};

const TOUR_STEP_ORDER = [
    'dataStructures',
    'codeAndNotes',
    'createMenu',
    'algorithms',

// ========== TOUR ENGINE (LOCATION-INSENSITIVE DYNAMIC POSITIONING) ========== //

(function initWelcomeTour() {
    const spotlightEl = document.getElementById('tour-spotlight');
    const cardEl = document.getElementById('tour-card');
    const arrowEl = document.getElementById('tour-arrow');
    const titleEl = document.getElementById('tour-card-title');
    const counterEl = document.getElementById('tour-step-counter');
    const descEl = document.getElementById('tour-card-desc');
    const skipBtn = document.getElementById('tour-skip-btn');
    const prevBtn = document.getElementById('tour-prev-btn');
    const nextBtn = document.getElementById('tour-next-btn');
    const btnHelp = document.getElementById('btnHelp');

    let activeStepIndex = 0;
    let isTourRunning = false;
    let wasZenModeBeforeTour = false;

    function isElementVisible(el) {
    if (!el) return false;
    const style = window.getComputedStyle(el);
    if (style.display === 'none' || style.visibility === 'hidden' || parseFloat(style.opacity) === 0) {
        return false;
    }
    const rect = el.getBoundingClientRect();
    return rect.width > 1 && rect.height > 1;
    }

    function resolveStepTargetElement(stepConfig) {
    const candidateIds = [stepConfig.id, ...(stepConfig.fallbackIds || [])];
    for (const id of candidateIds) {
        const el = document.getElementById(id);
        if (isElementVisible(el)) return el;
    }
    return null;
    }

    function positionTourAroundTarget(targetEl) {
    if (!targetEl || !isTourRunning) return;

    const rect = targetEl.getBoundingClientRect();
    const pad = 4;
    const vw = window.innerWidth;
    const vh = window.innerHeight;

    // 1. Position Spotlight Highlight Box
    spotlightEl.style.display = 'block';
    spotlightEl.style.top = `${Math.max(0, rect.top - pad)}px`;
    spotlightEl.style.left = `${Math.max(0, rect.left - pad)}px`;
    spotlightEl.style.width = `${Math.min(vw, rect.width + pad * 2)}px`;
    spotlightEl.style.height = `${Math.min(vh, rect.height + pad * 2)}px`;

    // 2. Measure Tour Card & Compute Best Placement Side
    const cardRect = cardEl.getBoundingClientRect();
    const cardW = cardRect.width || Math.min(320, vw - 24);
    const cardH = cardRect.height || 165;
    const gap = 12;
    const edgeMargin = 10;

    const spaceBottom = vh - rect.bottom;
    const spaceTop = rect.top;
    const spaceRight = vw - rect.right;
    const spaceLeft = rect.left;

    let placement = 'bottom';
    if (spaceBottom >= cardH + gap) {
        placement = 'bottom';
    } else if (spaceTop >= cardH + gap) {
        placement = 'top';
    } else if (spaceRight >= cardW + gap) {
        placement = 'right';
    } else if (spaceLeft >= cardW + gap) {
        placement = 'left';
    } else {
        placement = spaceBottom >= spaceTop ? 'bottom' : 'top';
    }

    const targetCenterX = rect.left + rect.width / 2;
    const targetCenterY = rect.top + rect.height / 2;
    let top = 0;
    let left = 0;

    if (placement === 'bottom') {
        top = rect.bottom + gap;
        left = targetCenterX - cardW / 2;
    } else if (placement === 'top') {
        top = rect.top - cardH - gap;
        left = targetCenterX - cardW / 2;
    } else if (placement === 'right') {
        left = rect.right + gap;
        top = targetCenterY - cardH / 2;
    } else if (placement === 'left') {
        left = rect.left - cardW - gap;
        top = targetCenterY - cardH / 2;
    }

    // Clamp card coordinates so it never overflows the screen on mobile
    left = Math.max(edgeMargin, Math.min(vw - cardW - edgeMargin, left));
    top = Math.max(edgeMargin, Math.min(vh - cardH - edgeMargin, top));

    cardEl.style.left = `${left}px`;
    cardEl.style.top = `${top}px`;

    // 3. Position the Pointer Arrow toward the target's center
    arrowEl.style.top = '';
    arrowEl.style.bottom = '';
    arrowEl.style.left = '';
    arrowEl.style.right = '';
    arrowEl.style.borderTop = 'none';
    arrowEl.style.borderLeft = 'none';
    arrowEl.style.borderRight = 'none';
    arrowEl.style.borderBottom = 'none';

    const arrowHalf = 6;
    const arrowOffsetLimitX = Math.max(16, Math.min(cardW - 22, targetCenterX - left - arrowHalf));
    const arrowOffsetLimitY = Math.max(16, Math.min(cardH - 22, targetCenterY - top - arrowHalf));

    if (placement === 'bottom') {
        arrowEl.style.top = '-0.38rem';
        arrowEl.style.left = `${arrowOffsetLimitX}px`;
        arrowEl.style.borderTop = '0.0625rem solid var(--accent-primary)';
        arrowEl.style.borderLeft = '0.0625rem solid var(--accent-primary)';
    } else if (placement === 'top') {
        arrowEl.style.bottom = '-0.38rem';
        arrowEl.style.left = `${arrowOffsetLimitX}px`;
        arrowEl.style.borderBottom = '0.0625rem solid var(--accent-primary)';
        arrowEl.style.borderRight = '0.0625rem solid var(--accent-primary)';
    } else if (placement === 'right') {
        arrowEl.style.left = '-0.38rem';
        arrowEl.style.top = `${arrowOffsetLimitY}px`;
        arrowEl.style.borderBottom = '0.0625rem solid var(--accent-primary)';
        arrowEl.style.borderLeft = '0.0625rem solid var(--accent-primary)';
    } else if (placement === 'left') {
        arrowEl.style.right = '-0.38rem';
        arrowEl.style.top = `${arrowOffsetLimitY}px`;
        arrowEl.style.borderTop = '0.0625rem solid var(--accent-primary)';
        arrowEl.style.borderRight = '0.0625rem solid var(--accent-primary)';
    }
    }

    function showStep(index, direction = 1) {
    if (index < 0 || index >= TOUR_STEP_ORDER.length) {
        endTour(true);
        return;
    }

    activeStepIndex = index;
    const stepKey = TOUR_STEP_ORDER[activeStepIndex];
    const stepConfig = TOUR_STEPS_REGISTRY[stepKey];

    if (!stepConfig) {
        showStep(index + direction, direction);
        return;
    }

    if (typeof stepConfig.onEnter === 'function') {
        stepConfig.onEnter();
    }

    // Wait a frame so any DOM layout/modal changes from `onEnter` settle before measuring
    requestAnimationFrame(() => {
        const targetEl = resolveStepTargetElement(stepConfig);
        if (!targetEl) {
        showStep(index + direction, direction);
        return;
        }

        titleEl.textContent = stepConfig.title;
        descEl.textContent = stepConfig.description;
        counterEl.textContent = `${activeStepIndex + 1} / ${TOUR_STEP_ORDER.length}`;

        prevBtn.style.display = activeStepIndex === 0 ? 'none' : 'inline-flex';
        nextBtn.textContent = activeStepIndex === TOUR_STEP_ORDER.length - 1 ? 'Finish' : 'Next';

        cardEl.classList.add('open');
        requestAnimationFrame(() => positionTourAroundTarget(targetEl));
    });
    }

    function refreshCurrentStepPosition() {
    if (!isTourRunning) return;
    const stepKey = TOUR_STEP_ORDER[activeStepIndex];
    const stepConfig = TOUR_STEPS_REGISTRY[stepKey];
    if (!stepConfig) return;
    if (typeof stepConfig.onEnter === 'function') {
        stepConfig.onEnter();
    }
    requestAnimationFrame(() => {
        const targetEl = resolveStepTargetElement(stepConfig);
        if (targetEl) positionTourAroundTarget(targetEl);
    });
    }

    function endTour(markCompleted = true) {
    isTourRunning = false;
    cardEl.classList.remove('open');
    spotlightEl.style.display = 'none';

    // Close any modals opened during the tour and restore left mobile header tools
    tourHelpers.setDsaModal(false);
    tourHelpers.setMobileHistoryModal(false);
    tourHelpers.expandHeaderGroup('left');

    if (wasZenModeBeforeTour) {
        document.body.classList.add('zen-mode');
    }

    if (markCompleted) {
        try {
        localStorage.setItem(TOUR_STORAGE_KEY, 'true');
        } catch (e) { }
    }
    }

    function startTour(force = false) {
    try {
        if (!force && localStorage.getItem(TOUR_STORAGE_KEY) === 'true') {
        return;
        }
    } catch (e) { }

    wasZenModeBeforeTour = document.body.classList.contains('zen-mode');
    if (wasZenModeBeforeTour) {
        document.body.classList.remove('zen-mode');
    }

    isTourRunning = true;
    showStep(0, 1);
    }

    nextBtn.addEventListener('click', () => {
    if (activeStepIndex < TOUR_STEP_ORDER.length - 1) {
        showStep(activeStepIndex + 1, 1);
    } else {
        endTour(true);
    }
    });

    prevBtn.addEventListener('click', () => {
    if (activeStepIndex > 0) {
        showStep(activeStepIndex - 1, -1);
    }
    });

    skipBtn.addEventListener('click', () => endTour(true));

    if (btnHelp) {
    btnHelp.addEventListener('click', () => {
        startTour(true);
    });
    }

    window.addEventListener('resize', refreshCurrentStepPosition);
    window.addEventListener('orientationchange', () => setTimeout(refreshCurrentStepPosition, 150));
    window.addEventListener('mousemove', () => {
    if (isTourRunning && document.querySelector('.dragging')) {
        refreshCurrentStepPosition();
    }
    });
    window.addEventListener('touchmove', () => {
    if (isTourRunning && document.querySelector('.dragging')) {
        refreshCurrentStepPosition();
    }
    }, { passive: true });

    window.startShardSpiralTour = startTour;

    if (document.readyState === 'complete') {
    setTimeout(() => startTour(false), 450);
    } else {
    window.addEventListener('load', () => {
        setTimeout(() => startTour(false), 450);
    });
    }
})();
</script>
